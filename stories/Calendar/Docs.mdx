# Calendar

A Calendar component that can represent days, weeks and months. It can run on three different modes; single, multiple and range. These reflect the dates that can be selected. The implementation, alongside all the styling and props, may be found at [React Day Picker](https://react-day-picker.js.org/). Below we outline several attributes we think might be most used.

There are no extensions from React Day Pickers props to our own. In this sense, it is recommended to rely on the React Day Picker library for up to date information on the component. However it is recommended for a user to disable `showOutsideDays` for the range mode.

## Calendar

```tsx
'use client'

import * as React from 'react'
import { Chevron, DayPicker } from 'react-day-picker'
import { cn } from '@/lib/utils'
import { BsArrowLeft, BsArrowRight } from '@/assets'
import { cva } from 'class-variance-authority'
export type CalendarProps = React.ComponentProps<typeof DayPicker>

/**
 * Keys match react-day-picker v9's `UI`, `DayFlag` and `SelectionState` enums.
 * v9 moved the day modifiers (`selected`, `outside`, the range states, ...)
 * onto the `<td>` that `day` styles, and introduced `day_button` for the
 * `<button>` inside it — so the selection surface lives on `day`'s element and
 * `day_button` only carries sizing, typography and focus.
 */
const calendarVariants = cva([], {
  variants: {
    component: {
      months: ['relative', 'flex', 'flex-row', 'gap-4'],
      month: ['flex', 'w-full', 'flex-col', 'gap-2'],
      month_caption: ['flex', 'h-7', 'items-center', 'justify-center'],
      caption_label: ['text-sm', 'font-medium'],
      // v9 renders `nav` as a sibling of the months rather than inside the
      // caption, so it is positioned over the caption row of `months` here.
      nav: [
        'absolute',
        'inset-x-0',
        'top-0',
        'z-10',
        'flex',
        'h-7',
        'items-center',
        'justify-between',
        'bg-transparent',
        'dark:text-foreground-muted-dark'
      ],
      nav_button: [
        'flex',
        'h-7',
        'w-7',
        'items-center',
        'justify-center',
        'rounded-lg',
        'bg-transparent',
        'p-0',
        'outline-hidden',
        'focus-visible:shadow-blue',
        'aria-disabled:opacity-50'
      ],
      weekday: [
        'text-foreground-muted',
        'dark:text-foreground-muted-dark',
        'px-1',
        'py-2',
        'w-9',
        'font-semibold',
        'text-xs',
        'h-[34px]'
      ],
      day: [
        [
          'w-9',
          'p-0',
          'flex',
          'flex-col',
          'items-center',
          'text-center',
          'gap-1',
          'text-sm',
          'relative',
          'rounded-lg'
        ],
        ['focus-within:relative', 'focus-within:z-20']
      ],
      day_button: [
        'rounded-lg',
        'px-1',
        'w-full',
        'py-2',
        'h-[34px]',
        'flex',
        'flex-col',
        'justify-center',
        'text-sm',
        'font-bold',
        'items-center',
        'shrink-0',
        'focus-visible:shadow-blue',
        'outline-hidden',
        'disabled:cursor-not-allowed'
      ],
      selected: [
        'bg-accent',
        'text-foreground-inverse',
        'dark:bg-brandGreen-100',
        'dark:text-foreground-inverse-dark'
      ],
      outside: ['text-foreground-muted', 'dark:text-foreground-muted-dark'],
      disabled: [
        'text-foreground-muted',
        'dark:text-foreground-muted-dark',
        'opacity-50'
      ]
    }
  }
})

/**
 * Range days always carry `selected` as well, and both sets land on the same
 * `<td>`. `!` pins the range surface above `selected`'s rather than leaving the
 * winner to the order Tailwind happens to emit the two backgrounds in.
 */
const rangeEdge = [
  'bg-accent!',
  'text-foreground-inverse!',
  'dark:bg-blue-400!',
  'dark:text-foreground-inverse-dark!'
]

function Calendar({
  className,
  classNames,
  showOutsideDays = true,
  ...props
}: CalendarProps) {
  return (
    <DayPicker
      showOutsideDays={showOutsideDays}
      className={cn(
        'p-4 rounded-lg bg-background-alt dark:bg-background-alt-dark dark:text-foreground-inverse shadow-[0_4px_6px_0px_rgba(0,0,0,0.08)] flex w-fit',
        className
      )}
      classNames={{
        months: calendarVariants({ component: 'months' }),
        month: calendarVariants({ component: 'month' }),
        month_caption: calendarVariants({ component: 'month_caption' }),
        caption_label: calendarVariants({ component: 'caption_label' }),
        nav: calendarVariants({ component: 'nav' }),
        button_previous: calendarVariants({ component: 'nav_button' }),
        button_next: calendarVariants({ component: 'nav_button' }),
        month_grid: 'w-full border-collapse',
        weekdays: 'flex',
        weekday: calendarVariants({ component: 'weekday' }),
        week: 'flex w-full',
        day: calendarVariants({ component: 'day' }),
        day_button: calendarVariants({ component: 'day_button' }),
        selected: calendarVariants({ component: 'selected' }),
        range_start: cn(rangeEdge, 'rounded-l-full! rounded-r-none!'),
        range_end: cn(rangeEdge, 'rounded-r-full! rounded-l-none!'),
        range_middle:
          'rounded-none! bg-grey-400! text-grey-900! dark:bg-accent-alt! dark:text-foreground-inverse!',
        outside: calendarVariants({ component: 'outside' }),
        disabled: calendarVariants({ component: 'disabled' }),
        hidden: 'invisible',
        ...classNames
      }}
      components={{
        Chevron: props => {
          switch (props.orientation) {
            case 'left':
              return (
                <BsArrowLeft className="h-5 w-5 fill-current stroke-current" />
              )
            case 'right':
              return (
                <BsArrowRight className="h-5 w-5 fill-current stroke-current" />
              )
            default:
              return <Chevron {...props} />
          }
        }
      }}
      {...props}
    />
  )
}
Calendar.displayName = 'Calendar'

export { Calendar }
```
